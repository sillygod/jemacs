          "Toggle terminal copy mode together with Evil emacs/normal state."

        ;; ghostel has no debounce on the resize path: `ghostel--adjust-size' runs
        ;; from `window-size-change-functions', i.e. once per redisplay frame, and
        ;; each call resizes the PTY and redraws synchronously.  Dragging a divider
        ;; therefore sends dozens of SIGWINCHes.  A shell repaints one prompt line
        ;; and you never notice; a full-screen agent CLI repaints its entire UI
        ;; block and it strobes.  Collapse the drag into a single resize once it
        ;; settles.
        (defvar jemacs-ghostel-resize-debounce 0.15
          "Seconds of idle before a dragged ghostel window resizes its PTY.
Tuned when one resize meant one repaint.  A tmux-hosted agent --
ghostherd's tmux backend -- costs two: the agent redraws into tmux's
screen, and tmux then repaints the whole pane to the client, clear
included.  Raise it if resizing strobes.  The only cost of a larger
value is the terminal lagging behind the window while you drag.")

        (defvar jemacs-ghostel-resize-timers (make-hash-table :test 'eq)
          "Pending debounced ghostel resize timers, keyed by window.")

        (defun jemacs-ghostel-debounce-resize (orig window &optional force)
          "Defer ghostel's PTY resize until window dragging settles.
    FORCE paths -- font scaling, mode switches, explicit anchors -- are
    deliberately synchronous upstream, so they are passed straight through."
          (if force
              (funcall orig window force)
            (when-let* ((old (gethash window jemacs-ghostel-resize-timers)))
              (cancel-timer old))
            ;; Keyed per window: a single global timer would let two ghostel
            ;; windows cancel each other and leave one never resized.
            ;;
            ;; WINDOW and ORIG are passed as timer arguments rather than captured
            ;; in a closure.  Tangled config runs lexbound, but `C-c C-c' on an org
            ;; src block does not -- `org-babel-default-header-args:emacs-lisp'
            ;; defaults to (:lexical . "no") -- and a closure evaluated that way
            ;; loses both, failing later with a bare "void-variable window".
            (puthash window
                     (run-with-idle-timer
                      jemacs-ghostel-resize-debounce nil
                      (lambda (win fn)
                        (remhash win jemacs-ghostel-resize-timers)
                        (when (window-live-p win)
                          ;; force: repaint whatever went stale during the drag
                          (funcall fn win t)))
                      window orig)
                     jemacs-ghostel-resize-timers)))

        (use-package ghostel
          :ensure t
          :init
          ;; The native module lives outside straight's tree.  ghostel
          ;; downloads it into its own package directory by default, and a
          ;; rebuild of ghostel -- or a recursive one of a package that
          ;; needs it, like ghostherd -- deletes that directory, module and
          ;; all: the next start has none.  emacs-home/ is git-ignored.
          (setq ghostel-module-directory (locate-user-emacs-file "ghostel-module/"))
          :config
          (advice-add 'ghostel--adjust-size :around #'jemacs-ghostel-debounce-resize)
          (advice-add 'ghostel-copy-mode :after #'jemacs-ghostel-sync-evil-copy-mode)
          (advice-add 'ghostel-readonly-exit :after #'jemacs-ghostel-sync-evil-copy-mode)
          ;; Same reason as vterm: a terminal's keys belong to the program
          ;; in it.  Evil normal state eats j/k, Escape, C-v (Claude
          ;; image paste) and anything an agent TUI bound.  emacs state
          ;; passes them through; C-z toggles copy mode for motions.
          (define-key ghostel-mode-map (kbd "C-z") #'ghostel-copy-mode)
          (define-key ghostel-mode-map (kbd "C-c c") #'ghostel-project-cmd-run)
          (with-eval-after-load 'evil
            (evil-set-initial-state 'ghostel-mode 'emacs)
            (evil-define-key 'emacs 'ghostel-mode-map (kbd "C-z") #'ghostel-copy-mode)
            (evil-define-key 'normal 'ghostel-mode-map (kbd "C-z") #'ghostel-copy-mode)))

* project terminals: the dock

  A project's terminals live in =local/ghostel-dock/=: named ones
  (=*PROJECT:NAME*=, dev, test, shell...) in a bottom panel.  =SPC '=
  shows the panel with the terminal last used in the project, or hides
  it; =C-u SPC '= starts another, asking its name.  =SPC b t= lists
  every project's terminals in a ghostel-cmd style overlay: what each
  runs, its last exit status, its directory, how long since its output.

  The panel is kept per jworkspace layer.  Everything in jworkspace
  that snapshots a layer's windows or puts one back runs with the panel
  down (=ghostel-dock-without-panel=), so a layer never holds it in its
  layout; the dock shows each layer's own panel again after.  A
  snapshot that held it would bring it back unmarked, and maximize asks
  =one-window-p=, which a panel would answer.

  #+begin_src emacs-lisp
    (defun jemacs-ghostel-dock-layer ()
      "Name of the jworkspace layer shown: the dock keeps a panel per layer."
      (when-let* (((fboundp 'jworkspace--get-current-workspace))
                  (workspace (jworkspace--get-current-workspace)))
        (jworkspace-name workspace)))

    (defun jemacs-ghostel-dock-switch-layer (root)
      "Switch to ROOT's jworkspace layer before the dock opens a terminal of it.
    The layer is named after the project, as `consult-switch-project' names
    it; a new one is made only when asked."
      (when-let* ((layer (jemacs-project-layer-name root)))
        (when (or (jworkspace-worsapce-existp layer)
                  (y-or-n-p (format "create a new layer: %s" layer)))
          (jworkspace-switch-workspace layer))))

    (defun jemacs-ghostel-dock-setup (buffer caller)
      "Set up the new terminal BUFFER for the mode of CALLER, asked from."
      (when (buffer-live-p caller)
        (with-current-buffer buffer
          (project-ghostel-setup (buffer-local-value 'major-mode caller)))))

    (eval `(use-package ghostel-dock
             :straight (:local-repo ,(concat home-dir "local/ghostel-dock")

* search relevant functions

  Project search lives in =local/rg-panel/=: consult-ripgrep's
  minibuffer with the directory and filters in the prompt.  =M-o= opens
  the filter panel (globs, types, word/case/fixed/hidden, scope, rga,
  presets), =C-c C-e= edits the results in wgrep.  =SPC /=, or
  =SPC u SPC /= to pick the scope first (=C-u= scrolls in normal state).

  #+begin_src emacs-lisp
    (eval `(use-package rg-panel
             :straight (:local-repo ,(concat home-dir "local/rg-panel")
                                    :files ("rg-panel.el"))
             :commands (rg-panel-search)
             :custom
             (rg-panel-hint "M-o 篩選   C-c C-e 匯出成可編輯清單   SPC u 先選範圍")))

    (defun my-project-rg (&optional pick-scope)
      "Search the project with rg, the region, else the thing at point, to
    start with; outside a project, a directory asked for.  With a prefix
    argument PICK-SCOPE (SPC u), pick the scope first: the project, this
    file's directory, the uncommitted files or a directory.  In the
    search, M-o opens the filter panel and C-c C-e edits the results in
    place.  See `rg-panel-search'."
      (interactive "P")
      (let ((initial-input (cond
                            ((use-region-p) (buffer-substring-no-properties (region-beginning) (region-end)))
                            ((not (has-jlight-matches-p))
                             (buffer-substring-no-properties (car (get-pointer)) (cdr (get-pointer))))
                            (t nil))))
        (when (region-active-p)
          (deactivate-mark))
        (rg-panel-search initial-input pick-scope)))

  #+end_src
